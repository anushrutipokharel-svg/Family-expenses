// ── Auth ─────────────────────────────────────────────────── function isAuthed(token) { return TOKEN === "" || token === TOKEN; } function unauthorized() { return jsonResponse({ error: "unauthorized" }); } // ── Concurrency lock (prevents two writes clobbering each other) ── function withLock(fn) { const lock = LockService.getScriptLock(); try { lock.waitLock(12000); } catch (e) { return jsonResponse({ error: "busy" }); } } return sheet; } function normalizeKey(h) { if (h === "ID") return "id"; if (h === "OpeningBalance") return "openingBalance"; if (h === "AddedBy") return "addedBy"; return h.charAt(0).toLowerCase() + h.slice(1); break; case "budget": row = [ id, data.category || "Other", parseFloat(data.limit) || 0, data.addedBy || "", now ]; break; case "portfolio": row = [ id, data.type || "investment", data.name || "Item", parseFloat(data.amount) || 0, data.note || "", data.addedBy || "", now ];
